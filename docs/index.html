Quick startConnect, query, stream

main.go 
package main

import (
    "context"
    "fmt"
    "time"

    ibkr "github.com/shing1211/ibkrapi4go/pkg/ibkr"
)

func main() {
    cli, err := ibkr.NewClient(
        ibkr.WithGatewayURL("localhost:5001"),
        ibkr.WithTickleInterval(time.Hour),
    )
    if err != nil { /* handle */ }
    defer cli.Close()

    ctx := context.Background()
    cli.Session().Initialize(ctx)

    // List accounts
    accounts, _ := cli.Account().List(ctx)
    for _, a := range accounts {
        fmt.Println(a.ID, a.BaseCurrency)
    }

    // Stream live quotes via WebSocket
    sub, _ := cli.MarketData().SubscribeQuotes(ctx,
        ibkr.QuoteRequest{
            Contract:  ibkr.Stock("AAPL"),
            Fields:    []ibkr.Field{ibkr.FIELD_BID, ibkr.FIELD_ASK},
        },
    )
    defer sub.Close()

    for q := range sub.Events() {
        fmt.Printf("BID=%s  ASK=%s\n", q.Bid, q.Ask)
    }
}

OverviewWhat you get

A typed, ergonomic Go SDK for the Interactive Brokers Client Portal API and the IB REST API. Covers all 185 operations across both surfaces — account management, market data, orders, portfolio analytics, and more — with a channel-based streaming API, per-endpoint rate limiting, circuit breakers, and a built-in mock gateway for tests.

  • 185/185 CPAPI + REST operations
  • Channel-based WebSocket streaming
  • Context-aware with timeout propagation
  • Per-endpoint + global rate limiting
  • Circuit breaker per operation
  • Money as string (ADR-0008)
  • Dependency-free core
  • 47 fuzz targets + 180 shape tests
  • In-repo mock gateway
  • 47 benchmarks with CI regression gate
  • Structured slog logging
  • OTel-shaped metrics hooks
